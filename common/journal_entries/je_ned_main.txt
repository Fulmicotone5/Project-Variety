je_ned_main = {
	icon = "gfx/interface/icons/event_icons/event_skull.dds"
	group = je_group_historical_content
	scripted_button = je_ned_butt
	scripted_button = je_ned_butt_two
	timeout = 1000
	complete = {
		gold_reserves > 0
	}
	on_complete = {
		remove_modifier = modifier_ned_one
		if = {
				limit = {
					is_ai = yes
					OR = {
						c:BUD = {
						is_ai = yes
						is_subject = no
						}
						c:KBG = {
						is_ai = yes
						is_subject = no
						}
					}
					c:FLA = {
					is_ai = yes
					is_subject = no
					}
				}
				if = {
					limit = { 
						exists = c:BUD
					}
					annex_with_incorporation = c:BUD
				}
				if = {
					limit = { 
						exists = c:KBG
					}
					annex_with_incorporation = c:KBG
				}
				annex_with_incorporation = c:FLA
			}
	}
	on_timeout = {
		remove_modifier = modifier_ned_one
		if = {
			limit = { 
				is_ai = no
			}
			add_modifier = {
			name = modifier_ned_two
			months = 60
		}
		}
		hidden_effect = {
			if = {
				limit = { 
					is_ai = yes
				}
				trigger_event = {
					id = fm_ned.999
				}
			}
		}	
	}

	weight = 100
	should_be_pinned_by_default_uninvolved_or_context = yes
}
