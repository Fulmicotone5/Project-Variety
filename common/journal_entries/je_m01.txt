je_m01_main = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"
	group = je_group_historical_content
	timeout = 5351
	complete = {
		has_law = law_type:law_cultural_exclusion
		NOT = {
			has_law = law_type:law_serfdom
		}
		OR = {
			has_law = law_type:law_census_voting
			has_law = law_type:law_landed_voting
		}
		custom_tooltip = {
			text = coptocompletetool
			has_variable = coptocomplete
		}
		NOT = {
			any_military_formation = {
				filter = {
					is_army = yes
				}
				any_combat_unit = {
					OR = {
						has_unit_type = unit_type:combat_unit_type_hussars
						has_unit_type = unit_type:combat_unit_type_dragoons
						has_unit_type = unit_type:combat_unit_type_cuirassiers
						has_unit_type = unit_type:combat_unit_type_lancers
					}
					percent >= 0.25
				}
			}
		}
		any_military_formation = {
				filter = {
					is_army = yes
				}
				any_combat_unit = {
					OR = {
						has_unit_type = unit_type:combat_unit_type_cannon_artillery
						has_unit_type = unit_type:combat_unit_type_mobile_artillery
						has_unit_type = unit_type:combat_unit_type_shrapnel_artillery
						has_unit_type = unit_type:combat_unit_type_siege_artillery
					}
					percent >= 0.2
				}
			}
	}
	on_complete = {
		trigger_event = {
			id = fm_m01.2
		}
		remove_modifier = m01_mdf
	}
	fail = {
		any_scope_state = {
				is_capital = yes
				turmoil >= 0.5
			}
	}
	on_fail = {
		trigger_event = {
			id = fm_m01.3
		}
		remove_modifier = m01_mdf
	}
	on_timeout = {
		trigger_event = {
			id = fm_m01.3
		}
		remove_modifier = m01_mdf
	}
	should_be_pinned_by_default = yes
}

je_m01_abbasids = {
	icon = "gfx/interface/icons/religion_icons/sunni.dds"
	group = je_group_historical_content
	complete = {
		OR = {
			has_law = law_type:law_theocracy
			has_law = law_type:law_monarchy
		}
		owns_entire_state_region = STATE_DEIR_EZ_ZOR
		owns_entire_state_region = STATE_BAGHDAD
		owns_entire_state_region = STATE_MOSUL
		owns_entire_state_region = STATE_BASRA
		owns_entire_state_region = STATE_HAIL
		owns_entire_state_region = STATE_NEJD
		owns_entire_state_region = STATE_YEMEN
		owns_entire_state_region = STATE_ABU_DHABI
		owns_entire_state_region = STATE_OMAN
		any_scope_state = {
			count >= 6
			AND = {
				region = sr:region_persia
				owner = ROOT
			}
		}
	}
	fail = {
		ruler = {
			is_character_alive = no
		}
		NOT = {
			has_law = law_type:law_theocracy
		}
		NOT = {
			has_law = law_type:law_monarchy
		}
	}
	on_complete = {
		trigger_event = {
			id = fm_m01.5
		}
		capital = {
				convert_population = {
					target = root.religion
					value = 0.25
				}
			}
			every_scope_state = {
				limit = { is_capital = no }
				convert_population = {
					target = root.religion
					value = 0.15
				}				
			}
	}
	should_be_pinned_by_default = yes
}
