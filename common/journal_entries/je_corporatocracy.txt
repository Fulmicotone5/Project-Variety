	icon = "gfx/interface/icons/invention_icons/corporate_management.dds"

	group = je_group_internal_affairs
	immediate = {
		g:steel = {
			save_scope_as = corposteel2
		}
		set_variable = {
			name = steelbarfm
			value = 0
		}
	}
	progressbar = yes

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
	}
	fail = {
		NOT = {
			has_law_or_variant = law_type:law_corporatocracy
		}
	}

	on_complete = {
		remove_variable = steelbarfm
	}
	on_timeout = {
		add_modifier = {
			name = fmcorpo3
			days = 181
		}
		ig:ig_industrialists = {
				add_modifier = {
				name = modifier_corpoIG
				days = 181
			}
		}
	}
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					capital = {
					sg:steel = {
						AND = {
							import_tariff_level = max_tariffs
							export_tariff_level = max_subventions
							}
						}
					}
				}
				change_variable = {
					name = steelbarfm
					add = 1
				}
			}
		}
	}

	current_value = {
		value = var:steelbarfm
	}

	goal_add_value = {
		add = 6
	}

	timeout = 548

	weight = 200

	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
