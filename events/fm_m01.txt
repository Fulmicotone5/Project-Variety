fm_m01.3 = {
	# proclamation of the republic
	type = country_event
	placement = ROOT

	desc = fm_m01.3.d
	flavor = fm_m01.3.f
	duration = 3
	event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}
	on_opened_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_protest.dds"
	immediate = {
		set_variable = {
			name = lerevolution
			days = 300
		}
		activate_law = law_type:law_parliamentary_republic
		activate_law = law_type:law_landed_voting

		remove_primary_culture = cu:circassian
		ruler = {
			set_character_immortal = no

		}
		heir = {
			set_character_immortal = no
			kill_character = yes
		}
		ig:ig_intelligentsia = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_industrialists = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_trade_unions = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_landowners = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_rural_folk = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_armed_forces = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_devout = {
			leader = {
				kill_character = yes
			}
		}
		ig:ig_petty_bourgeoisie = {
			leader = {
				kill_character = yes
			}
		}
	}
	option = {
		name = fm_m01.3.a
		default_option = yes
		every_interest_group = {
			limit = {
				OR = {
					is_interest_group_type = ig_devout
					is_interest_group_type = ig_landowners
				}
			}
				add_modifier = {
					name = modifier_loth_ten
					days = normal_modifier_time
					is_decaying = yes
				}
			}
			every_interest_group = {
				limit = {
					is_interest_group_type = ig_armed_forces
				}
				add_modifier = {
					name = modifier_loth_eleven
					days = normal_modifier_time
					is_decaying = yes
				}
			}
			add_loyalists  = {
				strata = upper
				value = very_large_radicals
			}
			add_loyalists = {
				strata = middle
				value = large_radicals
			}
			add_loyalists = {
				strata = lower
				value = large_radicals
			}
		
	}
